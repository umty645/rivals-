local function CreateToggle(text, pos, callback)
    local btn = Instance.new("TextButton", MainFrame)
    btn.Size = UDim2.new(0, 210, 0, 55)
    btn.Position = pos
    btn.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
    btn.Text = text .. ": OFF"
    btn.TextColor3 = Color3.fromRGB(255, 80, 80)
    btn.Font = Enum.Font.SourceSansBold
    btn.TextSize = 14
    
    local state = false
    btn.MouseButton1Click:Connect(function()
        state = not state
        btn.Text = text .. (state and ": ON" or ": OFF")
        btn.TextColor3 = state and Color3.fromRGB(80, 255, 80) or Color3.fromRGB(255, 80, 80)
        callback(state)
    end)
end

-- [2. 타겟팅 시스템 (최적화)]
local function GetClosestTarget()
    local target, dist = nil, math.huge
    local Mid = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    for _, v in pairs(Players:GetPlayers()) do
        if v ~= LP and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
            if v.Team ~= LP.Team or v.Team == nil then
                local pos, vis = Camera:WorldToViewportPoint(v.Character.HumanoidRootPart.Position)
                if vis then
                    local mag = (Vector2.new(pos.X, pos.Y) - Mid).Magnitude
                    if mag < dist and mag < 800 then
                        target = v.Character.HumanoidRootPart
                        dist = mag
                    end
                end
            end
        end
    end
    return target
end

-- [3. 메인 로직 루프 (히트박스 & ESP)]
RS.RenderStepped:Connect(function()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            -- 히트박스 확장 (강제 고정)
            if _G.Hitbox and (p.Team ~= LP.Team or p.Team == nil) then
                for _, part in pairs(p.Character:GetChildren()) do
                    if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                        part.Size = Vector3.new(_G.Size, _G.Size, _G.Size)
                        part.Transparency = 0.8
                        part.CanCollide = false
                    end
                end
            end
            
            -- ESP (실루엣 고정)
            local hl = p.Character:FindFirstChild("Omni_Highlight")
            if _G.ESP and (p.Team ~= LP.Team or p.Team == nil) then
                if not hl then
                    hl = Instance.new("Highlight", p.Character)
                    hl.Name = "Omni_Highlight"
                    hl.FillTransparency = 1
                    hl.OutlineColor = Color3.fromRGB(255, 0, 0)
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                end
            elseif hl then hl:Destroy() end
        end
    end
end)

-- [4. 범용 월뱅 (Raycast + Meta Method Hooking)]
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
setreadonly(mt, false)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    -- 1. 월뱅 (Raycast 우회: 모든 장애물을 무시하고 타겟에게 직접 연결)
    if method == "Raycast" and _G.Wallbang and not checkcaller() then
        if args[3] then
            args[3].FilterDescendantsInstances = {workspace:FindFirstChild("Map"), workspace:FindFirstChild("Levels"), workspace.Terrain}
            args[3].FilterType = Enum.RaycastFilterType.Exclude
        end
    end

    -- 2. 타겟 강제 주입 (모든 FireServer 패킷 수정)
    if method == "FireServer" and _G.Wallbang and not checkcaller() then
        local t = GetClosestTarget()
        if t then
            for i, arg in pairs(args) do
                if typeof(arg) == "table" then
                    arg.HitPart = t
                    arg.HitPos = t.Position
                    if arg.Origin then arg.Origin = t.Position + Vector3.new(0, 5, 0) end
                elseif typeof(arg) == "Vector3" then
                    args[i] = t.Position
                end
            end
        end
    end
    
    if method == "Kick" then return nil end
    return oldNamecall(self, unpack(args))
end)
setreadonly(mt, true)
