-- Rivals Clean UI & FPS Boost Script
local lighting = game:GetService("Lighting")
local workspace = game:GetService("Workspace")
local players = game:GetService("Players")
local playerGui = players.LocalPlayer:WaitForChild("PlayerGui")

-- 1. 조명 및 그림자 제거 (회색 하늘 적용)
lighting.GlobalShadows = false
lighting.FogEnd = 9e9
lighting.Brightness = 1
for _, v in ipairs(lighting:GetChildren()) do
    if v:IsA("PostEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then
        v:Destroy()
    end
end

local sky = Instance.new("Sky")
sky.SkyboxBk = "rbxassetid://0"
sky.SkyboxDn = "rbxassetid://0"
sky.SkyboxFt = "rbxassetid://0"
sky.SkyboxLf = "rbxassetid://0"
sky.SkyboxRt = "rbxassetid://0"
sky.SkyboxUp = "rbxassetid://0"
sky.Parent = lighting

-- 2. 맵 구조물 매끄럽게 최적화 (그림자 및 불필요한 텍스처 제거)
for _, part in ipairs(workspace:GetDescendants()) do
    if part:IsA("BasePart") then
        part.Material = Enum.Material.SmoothPlastic
        part.Reflectance = 0
        part.CastShadow = false
    elseif part:IsA("Decal") or part:IsA("Texture") then
        part.Transparency = 1
    end
end

-- 3. UI 배경 및 박스 제거 (무기 아이콘 배경, 타이머 박스, 캐릭터 노란 배경 등)
task.spawn(function()
    while true do
        task.wait(1)
        pcall(function()
            for _, gui in ipairs(playerGui:GetDescendants()) do
                -- 이미지나 프레임 배경 중 아이콘 뒤의 회색/노란색/검은색 박스 투명화
                if gui:IsA("Frame") or gui:IsA("ImageLabel") or gui:IsA("ImageButton") then
                    -- 무기 슬롯이나 타이머, 프로필 영역의 배경 판정 제거
                    if gui.Name:lower():find("background") or gui.Name:lower():find("frame") or gui.Name:lower():find("box") then
                        gui.BackgroundTransparency = 1
                    end
                end
            end
        end)
    end
end)

print("Rivals Clean UI & FPS Boost Applied Successfully!")
