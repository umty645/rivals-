-- 3. 실시간 오브젝트 감지 및 적용 처리
local function applyTexturePack(item)
    if item:IsA("Sound") then
        for id, url in pairs(cdnMap) do
            if item.SoundId and item.SoundId:find(tostring(id)) then
                item.SoundId = url
            end
        end
        for id, _ in pairs(removeIds) do
            if item.SoundId and item.SoundId:find(tostring(id)) then
                item.Volume = 0
            end
        end
    elseif item:IsA("BasePart") or item:IsA("DataModelMesh") then
        for id, _ in pairs(removeIds) do
            -- ID 매칭되는 이펙트나 파츠 투명화 / 삭제
            pcall(function()
                item.Transparency = 1
                if item:IsA("BasePart") then
                    item.CanCollide = false
                end
            end)
        end
    end
end

Workspace.DescendantAdded:Connect(applyTexturePack)
PlayerGui.DescendantAdded:Connect(applyTexturePack)

for _, obj in ipairs(Workspace:GetDescendants()) do
    applyTexturePack(obj)
end

for _, obj in ipairs(PlayerGui:GetDescendants()) do
    applyTexturePack(obj)
end
