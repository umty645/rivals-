local function findMatchingAttachment(model, name)
	for _, child in ipairs(model:GetChildren()) do
		if child:IsA("Accessory") then continue end
		for _, att in ipairs(child:GetChildren()) do
			if att:IsA("Attachment") and att.Name == name then return att end
		end
		for _, grandchild in ipairs(child:GetChildren()) do
			if not grandchild:IsA("Accessory") then
				for _, att in ipairs(grandchild:GetChildren()) do
					if att:IsA("Attachment") and att.Name == name then return att end
				end
			end
		end
	end
end
local function weldAccessory(character, acc)
	acc.Parent = character
	local handle = acc:FindFirstChild("Handle")
	if not handle then warn("[Appearance] Accessory has no Handle:", acc.Name) return end

	for _, child in ipairs(handle:GetChildren()) do
		if child:IsA("Weld") or child:IsA("JointInstance") or child.Name == "AccessoryWeld" then
			child:Destroy()
		end
	end
	handle.Anchored = false

	local accAttachment = handle:FindFirstChildOfClass("Attachment")
	if accAttachment then
		local charAttachment = findMatchingAttachment(character, accAttachment.Name)
		if charAttachment then
			local weld = Instance.new("Weld")
			weld.Name = "AccessoryWeld"
			weld.Part0 = charAttachment.Parent
			weld.Part1 = handle
			weld.C0 = charAttachment.CFrame
			weld.C1 = accAttachment.CFrame
			weld.Parent = charAttachment.Parent
		else
			warn("[Appearance] No matching attachment for:", acc.Name, "->", accAttachment.Name)
		end
	else
		local head = character:FindFirstChild("Head")
		if head then
			local weld = Instance.new("Weld")
			weld.Name = "AccessoryWeld"
			weld.Part0 = head
			weld.Part1 = handle
			weld.C0 = CFrame.new(0, 0.5, 0)
			weld.C1 = acc.AttachmentPoint
			weld.Parent = head
		end
	end
end
local function stripCharacter(character)
	for _, v in ipairs(character:GetChildren()) do
		if v:IsA("Accessory") or v:IsA("Shirt") or v:IsA("Pants")
			or v:IsA("BodyColors") or v:IsA("ShirtGraphic") then

		end

end
function applyAppearance(character)
	if not previewModel then return end

	local template = previewModel:Clone()

	local upperTorso = character:WaitForChild("UpperTorso")
	character:WaitForChild("Humanoid")
	character:WaitForChild("HumanoidRootPart")

	task.wait(0.1)
	stripCharacter(character)

	task.wait()
	stripCharacter(character)

	local localHead = character:WaitForChild("Head")
	local previewHead = template:FindFirstChild("Head")
	if previewHead then
		local newHead = previewHead:Clone() 
		newHead.Name = "Head"
		newHead.CFrame = localHead.CFrame

		for _, child in ipairs(newHead:GetChildren()) do
			if child:IsA("Motor6D") or child:IsA("Weld") or child:IsA("JointInstance") then
				child:Destroy()
			end
		end

		local oldNeck = localHead:FindFirstChildOfClass("Motor6D")
		if oldNeck then
			oldNeck.Part1 = newHead
			oldNeck.Parent = newHead 
		else
			local upperNeck = upperTorso:FindFirstChild("Neck") or upperTorso:FindFirstChildOfClass("Motor6D")
			if upperNeck then
				upperNeck.Part1 = newHead
			end
		end

		localHead:Destroy()
		newHead.Parent = character
	end

	for _, v in ipairs(template:GetChildren()) do
		if v:IsA("Shirt") or v:IsA("Pants")
			or v:IsA("BodyColors") or v:IsA("ShirtGraphic") then
			v.Parent = character 
		end
	end

	for _, v in ipairs(template:GetChildren()) do
		if v:IsA("Accessory") then
			
			weldAccessory(character, v) 
		end
	end

	template:Destroy()

	local ourNames = {}
	for _, v in ipairs(previewModel:GetChildren()) do ourNames[v.Name] = true end

	local conn
	conn = character.ChildAdded:Connect(function(child)
		if child:IsA("Accessory") or child:IsA("Shirt") or child:IsA("Pants")
			or child:IsA("BodyColors") or child:IsA("ShirtGraphic") then
			if not ourNames[child.Name] then child:Destroy() end
		end
	end)

	character.AncestryChanged:Connect(function()
		if not character:IsDescendantOf(game) then
			conn:Disconnect()
		end
	end)
end

local function TrySpoofImage(img)
	if not img or not img:IsA("ImageLabel") then return end
	
	if (img.Image == TargetHeadshot ) then
		img.Image = NewHeadshot
	end
end
local function HookObject(obj)
	if connected[obj] then return end

	if obj:IsA("ImageLabel") then
		TrySpoofImage(obj)
		connected[obj] = obj:GetPropertyChangedSignal("Image"):Connect(function()
			TrySpoofImage(obj)
		end)
	end

	obj.Destroying:Connect(function()
		if connected[obj] then
			connected[obj]:Disconnect()
			connected[obj] = nil
		end
	end)
end
local function GetObject(path)
	local parts = string.split(path, ".")
	local ok, result = pcall(function()
		local node = (parts[1] == "Workspace") and workspace or game:GetService(parts[1])
		for i = 2, #parts do
			node = node:FindFirstChild(parts[i])
			if not node then return nil end
		end
		return node
	end)
	return ok and result or nil
end

local StatSpoof = {
	Wins = function(lbl) lbl.Text = Config.Wins end,
	WinPercent = function(lbl) lbl.Text = Config.WinPercent end,
}
local function TrySpoofLabel(lbl)
	if not lbl or not lbl:IsA("TextLabel") then return end

	local statFn = StatSpoof[lbl.Name]
	if statFn then statFn(lbl) return end

	local txt = lbl.Text
	if txt == "" then return end

	local newTxt = txt
	if _sameNameDisplay then
		
		newTxt = newTxt:gsub("@" .. namePat, Config.Username)
		newTxt = newTxt:gsub(namePat, Config.DisplayName)
	else
		
		newTxt = newTxt:gsub("@" .. namePat, Config.Username)
		newTxt = newTxt:gsub(displayPat, Config.DisplayName)
	end
	if newTxt ~= txt then lbl.Text = newTxt return end

	if MyRealLevel ~= "" and txt == MyRealLevel then
		lbl.Text = Config.Level lbl.Visible = true return
	end
	if MyRealStreak ~= "" and txt == MyRealStreak then
		lbl.Text = Config.Streak lbl.Visible = showStreak return
	end
end
local function TrySpoofContainer(obj)
	if not (obj:IsA("Frame") or obj:IsA("CanvasGroup")) then return end
	if obj.Name == "Streak" then
		local valLbl = obj:FindFirstChild("Value")
		if valLbl and valLbl:IsA("TextLabel") then
			valLbl.Text = Config.Streak
			valLbl.Visible = showStreak
		end
		obj.Visible = showStreak
	elseif obj.Name == "Level" then
		local valLbl = obj:FindFirstChild("Value") or obj:FindFirstChild("Title")
		if valLbl and valLbl:IsA("TextLabel") then
			if MyRealLevel == "" and valLbl.Text ~= "" and valLbl.Text ~= Config.Level then
				MyRealLevel = valLbl.Text
			end
			valLbl.Text = Config.Level
			valLbl.Visible = true
		end
		obj.Visible = not showStreak
	end
end
local function HookLabel(lbl)
	if connected[lbl] then return end
	TrySpoofLabel(lbl)
	connected[lbl] = lbl:GetPropertyChangedSignal("Text"):Connect(function()
		TrySpoofLabel(lbl)
	end)
	lbl.Destroying:Connect(function()
		if connected[lbl] then
			connected[lbl]:Disconnect()
			connected[lbl] = nil
		end
	end)
end
local function ScanDescendants(root)
	for _, obj in ipairs(root:GetDescendants()) do
		if connected[obj] then continue end 
		if obj:IsA("TextLabel") then HookLabel(obj)
		else TrySpoofContainer(obj) end
	end
end

local function FindRankedLeaderboard()
	local pg = LP:FindFirstChild("PlayerGui")
	if not pg then return nil end
	local target = Config.LeaderboardType or "Ranked"
	
	local escaped = target:gsub("[%^%$%(%)%%%.%[%]%*%+%-%?]", "%%%1")
	for _, child in ipairs(pg:GetChildren()) do
		local t = child:FindFirstChild("Title")
		if t and t:IsA("TextLabel") and t.Text:match("^%s*" .. escaped .. "%s*$") then
			return child
		end
	end
end

local function GetLeaderboardContainer()
	local lb = FindRankedLeaderboard()
	if not lb then return nil end
	local list = lb:FindFirstChild("List")
	if not list then return nil end
	return list:FindFirstChild("Container")
end

local function SpoofLeaderboardSlot(slot)
	if not Config.LeaderboardTargetRank or Config.LeaderboardTargetRank == "" then return end
	local left = slot:FindFirstChild("Left")
	if not left then return end

	local rankLabel = left:FindFirstChild("Rank")
	if not rankLabel or not rankLabel:IsA("TextLabel") then return end
	if rankLabel.Text:match("^%s*(.-)%s*$") ~= Config.LeaderboardTargetRank then return end

	local usernameLabel = left:FindFirstChild("Username")
	if usernameLabel and usernameLabel:IsA("TextLabel") and usernameLabel.Text ~= Config.Username then
		usernameLabel.Text = Config.Username
	end

	local displayLabel = left:FindFirstChild("DisplayName")
	if displayLabel and displayLabel:IsA("TextLabel") and displayLabel.Text ~= Config.DisplayName then
		displayLabel.Text = Config.DisplayName
	end

	local headshot = left:FindFirstChild("Headshot")
	if headshot and headshot:IsA("ImageLabel") and headshot.Image ~= NewHeadshot then
		headshot.Image = NewHeadshot
	end
end

local function HookLeaderboardSlot(slot)
	if lbConnected[slot] then return end
	task.wait() 

	SpoofLeaderboardSlot(slot)

	local left = slot:FindFirstChild("Left")
	if left then
		local conns = {}

		local rankLabel = left:FindFirstChild("Rank")
		if rankLabel then
			table.insert(conns, rankLabel:GetPropertyChangedSignal("Text"):Connect(function()
				SpoofLeaderboardSlot(slot)
			end))
		end

		local usernameLabel = left:FindFirstChild("Username")
		if usernameLabel then
			table.insert(conns, usernameLabel:GetPropertyChangedSignal("Text"):Connect(function()
				SpoofLeaderboardSlot(slot)
			end))
		end

		local displayLabel = left:FindFirstChild("DisplayName")
		if displayLabel then
			table.insert(conns, displayLabel:GetPropertyChangedSignal("Text"):Connect(function()
				SpoofLeaderboardSlot(slot)
			end))
		end

		local headshot = left:FindFirstChild("Headshot")
		if headshot then
			table.insert(conns, headshot:GetPropertyChangedSignal("Image"):Connect(function()
				SpoofLeaderboardSlot(slot)
			end))
		end

		lbConnected[slot] = conns
	end

	slot.Destroying:Connect(function()
		if lbConnected[slot] then
			for _, c in ipairs(lbConnected[slot]) do c:Disconnect() end
			lbConnected[slot] = nil
		end
	end)
end

local function WatchLeaderboard()
	if not Config.LeaderboardTargetRank or Config.LeaderboardTargetRank == "" then return end
	task.spawn(function()
		local containerConn = nil
		local lastContainer = nil
		while true do
			local container = GetLeaderboardContainer()
			if container and container ~= lastContainer then
				
				for _, slot in ipairs(container:GetChildren()) do
					HookLeaderboardSlot(slot)
				end
				
				if containerConn then containerConn:Disconnect() end
				containerConn = container.ChildAdded:Connect(HookLeaderboardSlot)
				lastContainer = container
			end
			
			if lastContainer then
				for _, slot in ipairs(lastContainer:GetChildren()) do
					SpoofLeaderboardSlot(slot)
				end
			end
			task.wait(2)
		end
	end)
end

	if not _cgInnerFrame then return nil end

	if not _cgPeopleList or not _cgPeopleList.Parent then
		local pPage = _cgInnerFrame:FindFirstChild("peoplepage")
		_cgPeopleList = pPage and pPage:FindFirstChild("PeopleList", true)
	end

	return _cgPeopleList
end

local function ApplyCoreGuiPlayerList()
	local list = _cgResolvePath()
	if not list then return end

	if not _cgCurrentRow or not _cgCurrentRow.Parent then
		_cgCurrentRow, _cgNameLabel, _cgDisplayNameLabel, _cgIcon = nil, nil, nil, nil
		for _, row in ipairs(list:GetChildren()) do
			local userLabel = row:FindFirstChild("UserName", true)
			if userLabel and userLabel:IsA("TextLabel") and (userLabel.Text == RealName or userLabel.Text == "@" .. RealName) then
				_cgCurrentRow = row
				_cgNameLabel = userLabel
				_cgDisplayNameLabel = row:FindFirstChild("DisplayName", true)
				_cgIcon = row:FindFirstChild("Avatar", true)
				if _cgIcon and not _cgIcon:IsA("ImageLabel") then
					_cgIcon = _cgIcon:FindFirstChildOfClass("ImageLabel") or _cgIcon
				end
				break
			end
		end
	end

	if _cgNameLabel and _cgNameLabel.Text ~= Config.Username then
		_cgNameLabel.Text = Config.Username
	end
	if _cgDisplayNameLabel and _cgDisplayNameLabel.Text ~= Config.DisplayName then
		_cgDisplayNameLabel.Text = Config.DisplayName
	end
	if _cgIcon and _cgIcon:IsA("ImageLabel") and _cgIcon.Image ~= _cgFakeIcon then
		_cgIcon.Image = _cgFakeIcon
	end
end

local function deepFind(root, ...)
	local cur = root
	for _, name in ipairs({...}) do
		cur = cur:FindFirstChild(name)
		if not cur then return nil end
	end
	return cur
end
